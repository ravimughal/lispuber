NGINX:
    gcloud compute machine-images create nginx-image --source-instance=nginx1-vm --zone=europe-southwest1-c

    gcloud compute instances delete nginx1-vm --zone=europe-southwest1-c --quiet

    gcloud compute instances create nginx1-vm --source-machine-image=nginx-image --private-network-ip=10.204.0.11   -
    -no-address --zone=europe-southwest1-c

    sudo nano /etc/nginx/conf.d/backend.conf:
        upstream backend {
            server 10.204.0.21:8080 max_fails=1 fail_timeout=60s;
            server 10.204.0.22:8080 max_fails=1 fail_timeout=60s;
            server 10.204.0.23:8080 backup;
            server 10.204.0.24:8080 backup;
            server 10.204.0.25:8080 backup;
        }

    sudo nano /etc/nginx/conf.d/backend-map.conf:
        map $upstream_addr $backend_name {
            default unknown;
            10.204.0.21:8080 app1-vm;
            10.204.0.22:8080 app2-vm;
            10.204.0.23:8080 app3-vm;
            10.204.0.24:8080 app4-vm;
            10.204.0.25:8080 app5-vm;
        }
    
    sudo nano /etc/nginx/sites-available/default:
        server {
            listen 80;
            server_name lispuber.xyz;
            return 301 https://$host$request_uri;
        }

        server {
            listen 443 ssl;
            server_name lispuber.xyz;
            root /home/apa/nginx;
            index index.html;

            # SSL
            ssl_certificate /etc/letsencrypt/live/lispuber.xyz/fullchain.pem;
            ssl_certificate_key /etc/letsencrypt/live/lispuber.xyz/privkey.pem;
            include /etc/letsencrypt/options-ssl-nginx.conf;
            ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

            # Assets
            location /js/ {
                root /home/apa/nginx;
                try_files $uri =404;
            }
            location /images/ {
                root /home/apa/nginx;
                try_files $uri =404;
            }
            location /css/ {
                root /home/apa/nginx;
                try_files $uri =404;
            }

            # API
            location /api/ {
                proxy_pass http://backend;
                proxy_http_version 1.1;
                proxy_set_header Host $host;
                proxy_set_header X-Real-IP $remote_addr;
                proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
                proxy_set_header X-Forwarded-Proto $scheme;

                proxy_connect_timeout 1s;
                proxy_read_timeout 3s;

                add_header X-Backend-Name $backend_name always;
            }

            # Languages
            location ~ ^/pt(/(.*))?$ {
                set $page $2;
                root /home/apa/nginx;
                try_files
                    /html/pt/$page.html
                    /html/pt/index.html
                    =404;
            }

            location ~ ^/en(/(.*))?$ {
                set $page $2;
                root /home/apa/nginx;
                try_files
                    /html/en/$page.html
                    /html/en/index.html
                    =404;
            }

            location / {
                set $page $uri;
                root /home/apa/nginx;
                try_files
                    /html/en/$page.html
                    /html/en/index.html
                    =404;
            }
        }
